namespace detail {

template <class Ntk,
          typename StaticParams = DefaultAreaResynStaticParams>
class AreaResynImpl {
  struct WindowManagerParams {
    static constexpr uint32_t max_num_leaves = StaticParams::kMaxNumLeaves;
    uint32_t max_num_divisors;
    uint32_t max_reconvergence;
    int32_t odc_levels;
  };

  struct SimulaManagerParams {
    uint32_t max_num_divisors = 128;
    uint32_t max_reconvergence = 10;
    int32_t odc_levels = 0;
    static constexpr uint32_t kMaxNumLeaves = StaticParams::kMaxNumLeavesXL;
    static constexpr uint32_t kNumVarsSigns = StaticParams::kNumVarsSign;
  };

  using Node = typename Ntk::node;
  using Signal = typename Ntk::signal;
  using ArrivalsTracker = analyzers::trackers::arrival_times_tracker<Ntk>;
  using RequiredTracker = analyzers::trackers::required_times_tracker<Ntk>;
  using Signature = kitty::static_truth_table<StaticParams::kMaxNumLeaves>;
  using SimulaManager = windowing::SimulaManager<Ntk, SimulaManagerParams>;
  using WindowManager = windowing::window_manager<Ntk, WindowManagerParams>;
  using WindowSimulator =
      windowing::window_simulator<Ntk, StaticParams::kMaxNumLeaves>;
  static constexpr uint32_t kMaxNumOutputs = Ntk::kMaxNumOutputs;

  struct SharedOptoWindow {
    WindowManager manager;
    WindowSimulator simulator;
    SimulaManager simula;
    Node root{};
  };

  struct LutSynthParams : synthesis::lut::DefaultLutSynthParams {
    static constexpr uint32_t kNumVarsCube = StaticParams::kMaxNumLeaves;
    static constexpr uint32_t kMaxInpsSolo = StaticParams::kMaxInpsSolo;
    static constexpr uint32_t kMaxInpsDual = StaticParams::kMaxInpsDual;
    static constexpr uint32_t kMaxNumGates = 32u;
  };

  using LutSynth =
      typename synthesis::lut::LutSynthDispatch<StaticParams::kLutSynth, Ntk,
                                                WindowManager, WindowSimulator,
                                                LutSynthParams>::type;

  using SpfdMeth = typename dependency::window::OracleDispatch<
      StaticParams::kSpfdMeth, Ntk, WindowManager, WindowSimulator,
      StaticParams::kMaxCutsSize>::type;

 public:
  AreaResynImpl(Ntk& ntk, AreaResynParams ps, AreaResynStats& st)
      : ntk_(ntk),
        ps_(ps),
        st_(st),
        wm_params_{ps_.max_num_divisors, ps_.max_reconvergence, ps_.odc_levels},
        sm_params_{ps_.max_num_divisors, ps_.max_reconvergence, ps_.odc_levels},
        window_{WindowManager(ntk_, wm_params_, wm_stats_),
                WindowSimulator(ntk_),
                SimulaManager(ntk_, sm_params_, sm_stats_)
              },
        synth_(ntk, window_.manager, window_.simulator, st_.st_synth),
        spfds_(ntk, window_.manager, window_.simulator),
        arrivals_(ntk_),
        required_(ntk_, 0) {
          required_.SetMaxDelay(arrivals_.worst_delay());

          for ( int n = 0; n < 9; ++n )
          {
            proj_fns_.emplace_back();
            for ( int i = 0; i < n; ++i ){
              proj_fns_[n].emplace_back( n );
              kitty::create_nth_var( proj_fns_[n][i], i );
            }
          }
        }

  void Run();
  void RunAsOrchestration();
  void RunAsFlow();
  void ReportNtkState();

 private:
  // === Phase decomposition ===
  void SetupTrackers();
  void OptimizeNode(Node);

  // === Orchestration (delegates to strategy classes) ===
  template <uint32_t NumOutputs>
  bool Orchestrate(Node);

  // === High-level optimization ===
  bool TryMergerSolo(Node);
  bool TryMODecoSolo(Node);
  bool Try0ResubSolo(Node);
  bool TryWResubSolo(Node);
  bool TryRewireSolo(Node);
  bool TryStructSolo(Node);
  bool TryWindowSolo(Node);
  bool TrySimulaSolo(Node);
  
  void FlowMergerSolo();
  void FlowMODecoSolo();
  void FlowRewireSolo();
  void Flow0ResubSolo();
  
  // === Shared helpers ===
  std::optional<Signal> Find0ResubSolo(Node);
  std::optional<Signal> FindBestDivSolo();
  std::optional<Signal> FindBestDiv(uint32_t);
  template <uint32_t kNumInputs>
  bool TrySynthesis(math::static_vector<Signal, kNumInputs> const&);
  bool ExistsPath(Node, Node);
  void Merge(Node, Node);
  bool SkipNode(Node);
  Signal Interpolate(Signature const&, std::set<Signal> const&);
 private:
  Ntk& ntk_;
  AreaResynParams ps_;
  AreaResynStats& st_;
  WindowManagerParams wm_params_;
  SimulaManagerParams sm_params_;
  windowing::window_manager_stats wm_stats_;
  windowing::SimulaManagerStats sm_stats_;
  SharedOptoWindow window_;
  LutSynth synth_;
  SpfdMeth spfds_;
  ArrivalsTracker arrivals_;
  RequiredTracker required_;
  std::vector<std::vector<kitty::dynamic_truth_table>> proj_fns_;

};

template <class Ntk, typename StaticParams>
void AreaResynImpl<Ntk, StaticParams>::ReportNtkState() {
  std::vector<uint32_t> mffc_sizes;
  ntk_.foreach_gate([&](auto const& n){
    window_.manager.run(n);
    uint32_t mffc_size = window_.manager.num_mffc();
    while (mffc_sizes.size()<=mffc_size)
      mffc_sizes.push_back(0);
    mffc_sizes[mffc_size]+=1;
    return true;
  });
  for ( uint32_t s=0; s < mffc_sizes.size(); ++s )
  {
    std::cout << s << " : " << mffc_sizes[s] << std::endl;
  }
}

void AreaResynImpl<Ntk, StaticParams>::Run() {
  //if (ps_.orchestration)

  //else
    RunAsFlow();
}

template <class Ntk, typename StaticParams>
void AreaResynImpl<Ntk, StaticParams>::RunAsOrchestration() {
  auto const size_init = ntk_.num_gates();
  ntk_.foreach_gate([&](auto const& n) {
    if (st_.GetNumOptos() >= ps_.stop_after_nth) return false;
    if (n > size_init || ntk_.is_dead(n) || ntk_.is_pi(n)) return true;
    if (ntk_.fanin_size(n) == 1) return true;

    OptimizeNode(n);
    return true;
  });
}

template <class Ntk, typename StaticParams>
void AreaResynImpl<Ntk, StaticParams>::RunAsFlow() {
  mockturtle::call_with_stopwatch( st_.time_merger_solo, [&]() { FlowMergerSolo(); } );
  mockturtle::call_with_stopwatch( st_.time_mo_dec_solo, [&]() { FlowMODecoSolo(); } );
}

template <class Ntk, typename StaticParams>
void AreaResynImpl<Ntk, StaticParams>::OptimizeNode(Node n) {
  window_.root = n;

  mockturtle::call_with_stopwatch(st_.time_window_constr,
                                  [&]() { window_.manager.run(n); });

  mockturtle::call_with_stopwatch(st_.time_window_simula, [&]() {
    window_.simulator.run(window_.manager); });

  if (ntk_.num_outputs(n) == 1)
    Orchestrate<1>(n);
  else if (ntk_.num_outputs(n) == 2)
    Orchestrate<2>(n);
  else {
    // gracefully skip instead of throwing
    fmt::print("[warn] Skipping node {} with {} outputs\n", n,
               ntk_.num_outputs(n));
  }
}

template <class Ntk, typename StaticParams>
template <uint32_t NumOutputs>
bool AreaResynImpl<Ntk, StaticParams>::Orchestrate(Node n) {
  if constexpr (NumOutputs == 1) {

    if (mockturtle::call_with_stopwatch(st_.time_merger_solo,
                                        [&]() { return TryMergerSolo(n); }))
    {
      st_.RecordMergerSolo();
      return true;
    }

    if ( !window_.manager.is_valid() ) return true;

    if (mockturtle::call_with_stopwatch(st_.time_rewire_solo,
                                        [&]() { return TryRewireSolo(n); }))
      return true;

    if (mockturtle::call_with_stopwatch(st_.time_0resub_solo,
                                        [&]() { return Try0ResubSolo(n); }))
      return true;

    if (mockturtle::call_with_stopwatch(st_.time_Wresub_solo,
                                        [&]() { return TryWResubSolo(n); }))
      return true;

    if (window_.manager.num_mffc() <= 1) return true;

    if (mockturtle::call_with_stopwatch(st_.time_struct_solo,
                                        [&]() { return TryStructSolo(n); }))
      return true;

    if (mockturtle::call_with_stopwatch(st_.time_window_solo,
                                        [&]() { return TryWindowSolo(n); }))
      return true;

    return true;
  } else if constexpr (NumOutputs == 2) {
    // TODO: dual-output strategy
    return true;
  } else {
    fmt::print("[warn] Unsupported {}-output node\n", NumOutputs);
    return false;
  }
}

template <class Ntk, typename StaticParams>
bool AreaResynImpl<Ntk, StaticParams>::Try0ResubSolo(Node n) {
  if (!ps_.try_0resub_solo) return false;
  std::optional<Signal> div = Find0ResubSolo(n);
  if (div) {
    st_.Record0ResubSolo();
    ntk_.substitute_node(n, *div);
    return true;
  }
  return false;
}

template <class Ntk, typename StaticParams>
std::optional<typename Ntk::signal>
AreaResynImpl<Ntk, StaticParams>::Find0ResubSolo(Node n) {
  assert(window_.root==n);
  Signal froot = ntk_.make_signal(n);
  auto const& signature_root = window_.simulator.Get(froot);
  std::optional<Signal> div;

  window_.manager.foreach_divisor([&](auto const& f, auto i) {
    if (div) return;
    if (ps_.preserve_times) {
      double const time_max = ps_.use_required
                                  ? required_.get_time(froot)
                                  : arrivals_.get_time(froot);
      if (arrivals_.get_time(f) > time_max) return;
    }
    auto const& signature_div = window_.simulator.Get(f);

    auto const diff = signature_div ^ signature_root;
    if (ps_.odc_levels <= 0) {
      auto const num_ones = kitty::count_ones(diff);
      if (num_ones == 0)
        div = f;
      else if (num_ones == diff.num_bits())
        div = ntk_.create_not(f);
    } else {
      auto const& care = window_.simulator.get_careset();
      auto const num_ones = kitty::count_ones(boolean::binary_and(diff, care));
      if (num_ones == 0)
        div = f;
      else if (num_ones == kitty::count_ones(care))
        div = ntk_.create_not(f);
    }
  });
  return div;
}

template <class Ntk, typename StaticParams>
bool AreaResynImpl<Ntk, StaticParams>::TryRewireSolo(Node n) {
  assert(window_.root==n);
  if (!ps_.try_rewire_solo) return false;
  if (ntk_.fanin_size(window_.root)==1) return false;

  Signal const froot = ntk_.make_signal(window_.root);
  auto const& care = window_.simulator.get_careset();
  auto const& func = window_.simulator.Get(froot);

  // Base per-fanin simulations by value (no dangling pointers)
  std::vector<Signature> sims_base;
  sims_base.reserve(ntk_.fanin_size(window_.root));
  ntk_.foreach_fanin(window_.root, [&](auto const& fi, auto) {
    sims_base.push_back(window_.simulator.Get(fi));
  });

  std::optional<std::pair<Signal, Signal>> rewire;

  ntk_.foreach_fanin(window_.root, [&](auto const& fi, auto ii) {
    if (rewire) return;

    auto const ni = ntk_.get_node(fi);
    // Candidate removal: non-PI, multi-fanin, ONLY fanout is the root
    bool removable = !ntk_.is_pi(ni) && (ntk_.fanin_size(ni) > 1);
    ntk_.foreach_fanout(
        ni, [&](auto const& no) { removable &= (no == window_.root); });
    if (!removable) return;

    // Create a modified copy with the ii-th input flipped
    auto sims_mod = sims_base;
    ntk_.foreach_fanin(window_.root, [&](auto const& fii, auto iii) {
      if (fii == fi) sims_mod[ii] = ~sims_mod[iii];
    });

    // Build a transient pointer view expected by compute()
    std::vector<Signature const*> sims_view;
    sims_view.reserve(sims_mod.size());
    for (auto& s : sims_mod) sims_view.push_back(&s);

    // Simulate root with flipped input ii
    Signature const sim_flip = ntk_.compute(froot, sims_view);

    // Care for this input: only minterms where flipping ii changes the root
    Signature const care_inp =
        boolean::binary_and(care, boolean::binary_xor(sim_flip, func));

    // Try to match input `fi` with a divisor under care_inp
    auto const& signature_inp = window_.simulator.Get(fi);

    window_.manager.foreach_divisor([&](auto const& d, auto) {
      if (rewire) return;

      if (ps_.preserve_times) {
        double const time_max = ps_.use_required
                                    ? required_.get_time(fi)
                                    : arrivals_.get_time(fi);
        if (arrivals_.get_time(d) > time_max) return;
      }

      auto const& signature_div = window_.simulator.Get(d);
      auto const diff = boolean::binary_xor(signature_div, signature_inp);
      auto const ones = kitty::count_ones(boolean::binary_and(diff, care_inp));

      if (ones == 0) {
        rewire = {fi, d};
      } else if (ones == kitty::count_ones(care_inp)) {
        rewire = {fi, ntk_.create_not(d)};
      }
    });
  });

  if (rewire) {
    // Rebuild root with one child replaced
    auto tt = ntk_.node_function(window_.root);
    auto children = ntk_.get_children(window_.root);
    for (auto& c : children) {
      if (c == rewire->first) {
        c = rewire->second;
      }
    }
    Signal const fnew = ntk_.create_node(children, tt);
    ntk_.substitute_node(window_.root, fnew);
    st_.RecordRewireSolo();
    return true;
  }
  return false;
}

template <class Ntk, typename StaticParams>
bool AreaResynImpl<Ntk, StaticParams>::TryStructSolo(Node n) {
  assert(window_.root==n);

  if (!ps_.try_struct_solo) return false;

  auto const& inps = window_.manager.get_inputs();

  if (TrySynthesis<WindowManager::inps_capacity>(inps)) {
    st_.RecordStructSolo();
    return true;
  }
  return false;
}

template <class Ntk, typename StaticParams>
template <uint32_t kNumInputs>
bool AreaResynImpl<Ntk, StaticParams>::TrySynthesis(
    math::static_vector<Signal, kNumInputs> const& inps) {
  Signal froot = ntk_.make_signal(window_.root);
  Signature const& func = window_.simulator.Get(froot);
  bool const synthesized =
      ps_.preserve_times
          ? synth_.template Run<kNumInputs>(func, inps, arrivals_)
          : synth_.template Run<kNumInputs>(func, inps);
  if (!synthesized) return false;
  if (synth_.NumGates() <= window_.manager.num_mffc()) {
    if (ps_.preserve_times) {
      double const time_max = ps_.use_required
                                  ? required_.get_time(froot)
                                  : arrivals_.get_time(froot);
      if (synth_.GetTime() > time_max) return false;
    }
    auto fnew = synth_.template Insert<kNumInputs>(inps);
    ntk_.substitute_node(window_.root, fnew);
    return true;
  }
  return false;
}

template <class Ntk, typename StaticParams>
bool AreaResynImpl<Ntk, StaticParams>::TryWindowSolo(Node n) {
  assert(window_.root==n);

  if (!ps_.try_window_solo) return false;
  spfds_.Init();
  auto cut = spfds_.GetNext();
  if (cut) {
    if (TrySynthesis<StaticParams::kMaxCutsSize>(*cut)) {
      st_.RecordWindowSolo();
      return true;
    }
  }

  return false;
}

template <class Ntk, typename StaticParams>
bool AreaResynImpl<Ntk, StaticParams>::TrySimulaSolo(Node n) {
  assert(window_.root==n);

  if (!ps_.try_simula_solo) return false;
  return false;
}

template <class Ntk, typename StaticParams>
bool AreaResynImpl<Ntk, StaticParams>::TryMergerSolo(Node n) {

    // Mark fanouts of n
    const auto color_fo = (n << 2u) | 2u;
    ntk_.foreach_fanout(n, [&](auto const& no) {
      ntk_.set_visited(no, color_fo);
    });

    // Collect inputs of n
    std::set<Signal> base_inputs;
    ntk_.foreach_fanin(n, [&](auto const& fi, auto) {
      base_inputs.insert(fi);
    });

    std::optional<Node> best_node;
    int best_cost = std::numeric_limits<int>::max();

    // Explore fanouts of fanins
    ntk_.foreach_fanin(n, [&](auto const& fi, auto) {
      ntk_.foreach_fanout(fi.index, [&](auto const& no) {
        if (n == no) return;
        if (ntk_.num_outputs(no) != 1u) return;
        if (ntk_.visited(no) == color_fo) return;

        // Compute combined inputs size
        std::set<Signal> merged_inputs = base_inputs;
        ntk_.foreach_fanin(no, [&](auto const& foi, auto) {
          merged_inputs.insert(foi);
        });

        if ( !ExistsPath(n, no) && ( (int)merged_inputs.size() < best_cost)) {
          //if ( !ps_.preserve_times || ( std::abs(arrival_.get_time( ntk_.make_signal( n ) ) - arrival_.get_time( ntk_.make_signal( no ) )) < 1 ) )
          {
            best_node = no;
            best_cost = (int)merged_inputs.size();
          }
        }
      });
    });

    if (best_node) {
      Merge(n, *best_node);
      return true;
    }
    return false;
}

template <class Ntk, typename StaticParams>
bool AreaResynImpl<Ntk, StaticParams>::TryMODecoSolo(Node n) {
  spfds_.Init();
  bool done = false;
  Signature target = window_.simulator.Get(ntk_.make_signal(n));

  while (!done)
  {
    auto d = FindBestDivSolo();
    if (!d) return false;
    auto const n1 = ntk_.get_node(*d);
    std::set<Signal> children;
    ntk_.foreach_fanin(ntk_.get_node(*d), [&](auto const& fi, auto ii){
      spfds_.Update(window_.simulator.Get(fi));
      children.insert(fi);
    });
    ntk_.incr_trav_id();
    window_.manager.MarkTfoInside(n1, ntk_.trav_id());
    while ((!spfds_.IsCovered())&&(children.size()<StaticParams::kMaxInpsDual))
    {
      auto d = FindBestDiv(ntk_.trav_id());
      if (!d) return false;
      children.insert(*d);
      spfds_.Update(window_.simulator.Get(*d));
    }
    if (spfds_.IsCovered())
    {
      auto fnew = Interpolate(target, children);
      ntk_.substitute_node( n, fnew );
      Merge(n1, ntk_.get_node(fnew));
      st_.RecordMODecoSolo();
    }
    done = true;
  }
  return false;
}

template <class Ntk, typename StaticParams>
bool AreaResynImpl<Ntk, StaticParams>::ExistsPath(Node n1, Node n2) {
  if (n1==n2) return true;
  Node n_upper = n1;
  Node n_lower = n2;
  Signal f_upper = ntk_.make_signal(n_upper);
  Signal f_lower = ntk_.make_signal(n_lower);
  double t_upper = arrivals_.get_time( f_upper );
  double t_lower = arrivals_.get_time( f_lower );

  if ( t_upper < t_lower )
  {
    std::swap( n_upper, n_lower );
    std::swap( f_upper, f_lower );
  }

  ntk_.incr_trav_id();
  bool found = false;
  std::function<void(Node)> DepthFirstSearch = [&](Node u) {
    if (found) return;
    if (u == n_lower) { found = true; return; }
    ntk_.foreach_fanin(u, [&](auto fi, auto) {
      auto v = ntk_.get_node(fi);
      if (ntk_.visited(v) != ntk_.trav_id() && ( arrivals_.get_time( fi ) >= (arrivals_.get_time(f_lower)*0.9) ))
      {
        ntk_.set_visited(v, ntk_.trav_id());
        DepthFirstSearch(v);
      }
    });
  };
  DepthFirstSearch(n_upper);
  return found;
}

template <class Ntk, typename StaticParams>
void AreaResynImpl<Ntk, StaticParams>::Merge(Node n1, Node n2) {
    if ( ntk_.is_pi( n1 ) || ntk_.is_pi( n2 ) ) return;
    auto tt1 = ntk_.node_function( n1 );
    auto tt2 = ntk_.node_function( n2 );

    std::unordered_map<uint64_t, std::pair<int, int>> map;
    std::set<Signal> inps;
    ntk_.foreach_fanin( n1, [&]( auto const& fi, auto ii ){ inps.insert( fi ); } );
    ntk_.foreach_fanin( n2, [&]( auto const& fi, auto ii ){ inps.insert( fi ); } );
    if ( inps.size() > StaticParams::kMaxInpsDual ) return;
    int i = 0;
    std::vector<Signal> children;
    for ( auto const& f : inps )
    {
      children.push_back( f );
      map[f.data] = std::pair{ inps.size(), i++ };
    }
    std::vector<kitty::dynamic_truth_table const*> sim_ptrs1;
    std::vector<kitty::dynamic_truth_table const*> sim_ptrs2;

    ntk_.foreach_fanin( n1, [&]( auto const& fi, auto ii ){
      auto const p = map[fi.data];
      sim_ptrs1.push_back( &proj_fns_[p.first][p.second] );
    } );
    auto funcs1 = ntk_.compute( n1, sim_ptrs1 );

    ntk_.foreach_fanin( n2, [&]( auto const& fi, auto ii ){
      auto const p = map[fi.data];
      sim_ptrs2.push_back( &proj_fns_[p.first][p.second] );
    } );
    auto funcs2 = ntk_.compute( n2, sim_ptrs2 );

    std::vector<kitty::dynamic_truth_table> tts = { funcs1[0], funcs2[0] };

    auto fnew = ntk_.create_node( children, tts );
    ntk_.substitute_node( n1, Signal{ fnew.index, 0 } );
    ntk_.substitute_node( n2, Signal{ fnew.index, 1 } );
}

template <class Ntk, typename StaticParams>
void AreaResynImpl<Ntk, StaticParams>::FlowMergerSolo()
{
  if (!ps_.try_merger_solo) return;

  const auto nmax = ntk_.size();
  int cnt = 0;

  ntk_.foreach_gate([&](auto const& n) {
    if ( ntk_.is_pi(n) ) return true;
    if (cnt++ >= nmax) return false;
    if (st_.GetNumOptos() >= ps_.stop_after_nth) return false;
    if (SkipNode(n)) return true;
    if (ntk_.num_outputs(n) != 1) return true;

    if( TryMergerSolo(n) )
      st_.RecordMergerSolo();

    return true;
  });   
}


template <class Ntk, typename StaticParams>
void AreaResynImpl<Ntk, StaticParams>::FlowRewireSolo()
{
  if (!ps_.try_rewire_solo) return;
  const auto nmax = ntk_.size();
  int cnt = 0;

  ntk_.foreach_gate([&](auto const& n) {
    if ( ntk_.is_pi(n) ) return true;
    if (cnt++ >= nmax) return false;
    if (st_.GetNumOptos() >= ps_.stop_after_nth) return false;
    if (SkipNode(n)) return true;
    if (ntk_.num_outputs(n) != 1) return true;

    mockturtle::call_with_stopwatch(st_.time_window_constr, [&]() { window_.manager.run(n); });
    mockturtle::call_with_stopwatch(st_.time_window_simula, [&]() { window_.simulator.run(window_.manager); });

    if( TryRewireSolo(n) )
      st_.RecordRewireSolo();

    return true;
  });   
}

  template <class Ntk, typename StaticParams>
  auto AreaResynImpl<Ntk, StaticParams>::FindBestDiv(uint32_t bad_visited) -> std::optional<Signal>
  {
    std::optional<Signal> best_sign;
    uint32_t best_cost = std::numeric_limits<uint32_t>::max();
    window_.manager.foreach_divisor( [&]( auto const& f, auto i ){
      auto n = ntk_.get_node(f);
      if (ntk_.visited(n)!=bad_visited)
      {
        uint32_t cost = spfds_.Evaluate( window_.simulator.Get(f) );
        if (cost<best_cost)
        {
          best_cost = cost;
          best_sign = std::make_optional(f);
        }
      }
    });
    return best_sign;
  }

  template <class Ntk, typename StaticParams>
  auto AreaResynImpl<Ntk, StaticParams>::Interpolate(Signature const& target, std::set<Signal> const& fanins) -> Signal
  {
    std::vector<Signal> children;
    for ( auto const& s : fanins ) children.push_back(s);
    uint32_t num_minterms = 1u << children.size();
    kitty::dynamic_truth_table func(children.size());
    Signature tmp;
    for ( uint32_t m = 0; m < num_minterms; ++m )
    {
      tmp |= ~tmp;
      for ( uint32_t v = 0; v < children.size(); ++v )
      {
        if ( ( ( m >> v ) & 0x1 ) > 0 )
          tmp &= window_.simulator.Get(children[v]);
        else
          tmp &= ~window_.simulator.Get(children[v]);
      }
      if ( kitty:: count_ones(target & tmp) > 0 )
        kitty::set_bit(func, m);
      else
        kitty::clear_bit(func, m);
    }
    return ntk_.create_node(children, func);
  }

  template <class Ntk, typename StaticParams>
  auto AreaResynImpl<Ntk, StaticParams>::FindBestDivSolo() -> std::optional<Signal>
  {
    std::optional<Signal> best_sign;
    uint32_t best_cost = std::numeric_limits<uint32_t>::max();
    window_.manager.foreach_divisor( [&]( auto const& f, auto i ){
      auto n = ntk_.get_node(f);
      if ( !window_.manager.is_input(n) && (ntk_.num_outputs(n) == 1)&& (ntk_.fanin_size(n) <= StaticParams::kMaxInpsDual) )
      {
        uint32_t cost = spfds_.Evaluate( window_.simulator.Get(f) );
        if (cost<best_cost)
        {
          best_cost = cost;
          best_sign = std::make_optional(f);
        }
      }
    });
    return best_sign;
  }

  template <class Ntk, typename StaticParams>
  bool AreaResynImpl<Ntk, StaticParams>::SkipNode( Node n )
  {
    if ( ntk_.fanout_size( n ) <= 0 )
      return true;
    if ( ntk_.is_pi( n ) || ntk_.is_constant( n ) )
      return true;
    if ( ntk_.is_dead( n ) )
      return true;
    return false;
  }
}

// Public API
template <class Ntk,
          typename StaticParams = DefaultAreaResynStaticParams>
void AreaResyn(Ntk& ntk, AreaResynParams ps = {}, AreaResynStats* pst = nullptr) {
  AreaResynStats st;
  detail::AreaResynImpl<Ntk, StaticParams> p(ntk, ps, st);
  p.ReportNtkState();
  p.Run();
  p.ReportNtkState();
  if (pst) *pst = st;
}
